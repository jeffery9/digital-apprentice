## C.9 长时程上线门禁四项表（第 16 章 16.9）

□ **可停**：任意时刻 kill -9 后重启，验证通过才打点的检查点保证不重做已完成步。（演练记录：日期＿＿）
□ **可看**：进度事件流能回答"做到哪了/卡在哪/预计何时完"，且字段与合同承诺一致。
□ **可算**：跨天累计预算护栏生效，烧穿自动冻结并告警（不许只在报表里看见）。
□ **可交**：任意时点生成合格交接文档，人类接手可在 30 分钟内继续推进。
四项缺一不得上生产；每项附最近一次演练日期，过期一季度视同未过。

## C.16 TypeScript 对照片段（决策 #3：只给对照，不给完整 demo）

以下片段对应各章 Python demo 的核心接口形态，供 TS 栈团队评估移植成本。API 名称为示意，落地时以当期框架官方类型为准【L3 锚点｜截至 2026-10】。

```typescript
// 对照 D-01：最小 agent 循环 —— 终止条件是一等公民
type Step = { thought: string; action?: ToolCall; observation?: string };

async function agentLoop(init: Context, budget: Budget): Promise<Trajectory> {
  const traj: Step[] = [];
  while (budget.remaining() > 0) {                    // 预算护栏进类型签名
    const step = await decide(init, traj);            // 结构化输出守卫：返回即校验
    if (step.action == null) break;                   // 显式终态，不靠"看起来完成了"
    step.observation = await execute(step.action);    // 副作用隔离：写操作带幂等键
    traj.push(step);
    if (await guard.rejects(traj)) break;             // 验证通过才打点（第 10 章口径）
  }
  return freeze(traj);                                // 轨迹是审计对象，返回前封口
}
```

// 对照 D-03/D-04：能力画像路由 + 快照钉扎
interface CapabilityProfile {
  readonly modelId: string;
  readonly pinnedVersion: string;        // 不接受 "latest"
  readonly structuredOutputPass: number; // 自有探针实测值
  readonly costPerMTokIn: number;
  readonly residency: 'eu' | 'cn' | 'global';
  readonly measuredAt: Date;             // 画像有过期日
}

function route(task: TaskClass, pool: CapabilityProfile[]): CapabilityProfile {
  const eligible = pool.filter(p => p.meets(task) && !p.expired());
  if (eligible.length === 0) throw new NoCapacityError(task);   // 宁可拒绝，不可静默升舱
  return cheapestSufficient(eligible, task.qualityFloor);       // 级联升级在此之后
}
